using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Audio;

namespace GameAudioDeprecated
{

    public class AudioManager : MonoBehaviour
    {
        public static AudioManager Instance { get; private set; }

        [Header("Mixer")]
        public AudioMixer mixer;

        [System.Serializable]
        public class CategoryBinding
        {
            public SoundCategory category;
            public AudioMixerGroup mixerGroup;
            [Tooltip("Exposed float param name on the mixer, in dB, e.g. 'SFXVolume'. Leave empty if this category shouldn't have its own slider.")]
            public string exposedVolumeParam;
        }

        [Header("Category -> Mixer Group bindings")]
        [Tooltip("Add one entry per SoundCategory (except Music, which MusicManager routes itself).")]
        public List<CategoryBinding> categoryBindings = new List<CategoryBinding>();

        [System.Serializable]
        public class StageEnvironmentLoop
        {
            public string stageName;
            [Tooltip("The single ambience loop for this stage. Its Resource can be a plain AudioClip or an Audio Random Container, same as any SoundEvent.")]
            public SoundEvent environmentLoop;
        }

        [Header("Environment Loop (one per stage)")]
        [Tooltip("Index matches your stage numbering — keep this in the same order as MusicManager's Stages list. SetEnvironmentStage(index, ...) swaps to the matching entry and stops whatever was playing before.")]
        public List<StageEnvironmentLoop> environmentLoopsByStage = new List<StageEnvironmentLoop>();

        private const string EnvironmentLoopId = "EnvironmentStageLoop";
        private int _currentEnvironmentStage = -1;

        [Header("Pooling")]
        [SerializeField] private int initialPoolSize = 16;
        [SerializeField] private Transform poolParent;

        private readonly Queue<PooledAudioSource> _available = new Queue<PooledAudioSource>();
        private readonly List<PooledAudioSource> _active = new List<PooledAudioSource>();
        private readonly Dictionary<string, PooledAudioSource> _loops = new Dictionary<string, PooledAudioSource>();

        private void Awake()
        {
            // initializes AudioManager as a singleton
            if (Instance != null && Instance != this)
            {
                Destroy(gameObject);
                return;
            }
            Instance = this;
            DontDestroyOnLoad(gameObject);

            if (poolParent == null)
            {
                var go = new GameObject("PooledAudioSources");
                go.transform.SetParent(transform);
                poolParent = go.transform;
            }

            for (int i = 0; i < initialPoolSize; i++)
                _available.Enqueue(CreatePooledSource());
        }

        private void Start()
        {
            SetEnvironmentStage(0, Vector3.zero);
        }

        private PooledAudioSource CreatePooledSource()
        {
            var go = new GameObject("PooledAudioSource");
            go.transform.SetParent(poolParent);
            go.AddComponent<AudioSource>();
            var src = go.AddComponent<PooledAudioSource>();
            src.gameObject.SetActive(false);
            return src;
        }

        private PooledAudioSource GetFromPool()
        {
            PooledAudioSource src = _available.Count > 0 ? _available.Dequeue() : CreatePooledSource();
            src.gameObject.SetActive(true);
            _active.Add(src);
            return src;
        }

        private void ReturnToPool(PooledAudioSource src)
        {
            _active.Remove(src);
            if (!string.IsNullOrEmpty(src.LoopId))
                _loops.Remove(src.LoopId);
            src.transform.SetParent(poolParent);
            src.gameObject.SetActive(false);
            _available.Enqueue(src);
        }

        private AudioMixerGroup GetGroup(SoundCategory category)
        {
            foreach (var b in categoryBindings)
                if (b.category == category) return b.mixerGroup;
            return null;
        }

        private void EnsureGroup(SoundEvent evt)
        {
            if (evt.mixerGroup == null)
                evt.mixerGroup = GetGroup(evt.category);
        }

        // ---------------- Public API ----------------

        /// <summary>One-shot 3D sound at a fixed position (NPC/player hurt-death-attack, spell impact).</summary>
        public void PlaySFX(SoundEvent evt, Vector3 position)
        {
            if (evt == null || evt.GetClip() == null) return;
            EnsureGroup(evt);
            var src = GetFromPool();
            src.Play(evt, position, ReturnToPool);
        }

        /// <summary>One-shot 3D sound that follows a moving transform (spell projectile in flight).</summary>
        public void PlaySFXFollowing(SoundEvent evt, Transform target)
        {
            if (evt == null || evt.GetClip() == null || target == null) return;
            EnsureGroup(evt);
            var src = GetFromPool();
            src.PlayFollowing(evt, target, ReturnToPool);
        }

        /// <summary>
        /// Non-spatial UI sound (buttons, hover). Routed through the same pool as everything
        /// else — AudioSource.PlayOneShot doesn't accept an AudioResource, only a plain
        /// AudioClip, so it can't play Audio Random Container UI sounds. Make sure UI
        /// SoundEvents have Spatial Blend set to 0 so position doesn't matter.
        /// </summary>
        public void PlayUI(SoundEvent evt)
        {
            if (evt == null || evt.GetClip() == null) return;
            EnsureGroup(evt);
            var src = GetFromPool();
            src.Play(evt, transform.position, ReturnToPool);
        }

        /// <summary>
        /// Environment RFX scattered randomly around a center point (e.g. the player) —
        /// distant creaks, bird calls, monster growls in the fog, etc.
        /// </summary>
        public void PlayRandomAround(SoundEvent evt, Vector3 center, float minRadius, float maxRadius)
        {
            if (evt == null || evt.GetClip() == null) return;
            Vector2 dir = Random.insideUnitCircle.normalized;
            float dist = Random.Range(minRadius, maxRadius);
            Vector3 offset = new Vector3(dir.x, 0f, dir.y) * dist;
            PlaySFX(evt, center + offset);
        }

        /// <summary>Starts a looping sound (environment ambience loop) identified by loopId so it can be stopped later.</summary>
        public void PlayLoop(SoundEvent evt, Vector3 position, string loopId)
        {
            if (evt == null || evt.GetClip() == null || string.IsNullOrEmpty(loopId)) return;
            if (_loops.ContainsKey(loopId)) return; // already playing, ignore duplicate start
            EnsureGroup(evt);
            var src = GetFromPool();
            src.PlayLoop(evt, position, loopId);
            _loops[loopId] = src;
        }

        /// <summary>Same as PlayLoop, but the loop follows a moving transform (e.g. keep ambience centered on the player as they move).</summary>
        public void PlayLoopFollowing(SoundEvent evt, Transform target, string loopId)
        {
            if (evt == null || evt.GetClip() == null || target == null || string.IsNullOrEmpty(loopId)) return;
            if (_loops.ContainsKey(loopId)) return;
            EnsureGroup(evt);
            var src = GetFromPool();
            src.PlayLoopFollowing(evt, target, loopId);
            _loops[loopId] = src;
        }

        public void StopLoop(string loopId)
        {
            if (_loops.TryGetValue(loopId, out var src))
                src.StopImmediate();
        }

        public bool IsLoopPlaying(string loopId) => _loops.ContainsKey(loopId);

        /// <summary>
        /// Switches the single environment ambience loop to the entry assigned to this stage
        /// index in "Environment Loops By Stage", stopping whatever was playing before. A no-op
        /// if this stage is already the one playing. Call once when entering a new stage/area —
        /// e.g. alongside MusicManager.Instance.SetStage(index).
        /// </summary>
        public void SetEnvironmentStage(int stageIndex, Vector3 position)
        {
            if (stageIndex == _currentEnvironmentStage) return;
            StopLoop(EnvironmentLoopId);
            _currentEnvironmentStage = stageIndex;

            if (stageIndex < 0 || stageIndex >= environmentLoopsByStage.Count) return;
            var evt = environmentLoopsByStage[stageIndex].environmentLoop;
            if (evt == null) return;
            PlayLoop(evt, position, EnvironmentLoopId);
        }

        /// <summary>Same as SetEnvironmentStage, but the loop follows a moving transform (e.g. the player) instead of staying at a fixed point.</summary>
        public void SetEnvironmentStage(int stageIndex, Transform followTarget)
        {
            if (stageIndex == _currentEnvironmentStage) return;
            StopLoop(EnvironmentLoopId);
            _currentEnvironmentStage = stageIndex;

            if (stageIndex < 0 || stageIndex >= environmentLoopsByStage.Count) return;
            var evt = environmentLoopsByStage[stageIndex].environmentLoop;
            if (evt == null) return;
            PlayLoopFollowing(evt, followTarget, EnvironmentLoopId);
        }

        /// <summary>Stops the current environment stage loop, if any.</summary>
        public void StopEnvironmentLoop()
        {
            StopLoop(EnvironmentLoopId);
            _currentEnvironmentStage = -1;
        }

        public void StopAll()
        {
            foreach (var s in new List<PooledAudioSource>(_active))
                s.StopImmediate();
        }

        /// <summary>Sets a category's mixer volume from a linear 0..1 slider value (converted to dB).</summary>
        public void SetCategoryVolume(SoundCategory category, float linear01)
        {
            string param = null;
            foreach (var b in categoryBindings)
                if (b.category == category) { param = b.exposedVolumeParam; break; }
            if (string.IsNullOrEmpty(param) || mixer == null) return;

            float dB = linear01 <= 0.0001f ? -80f : Mathf.Log10(linear01) * 20f;
            mixer.SetFloat(param, dB);
        }
    }
}
